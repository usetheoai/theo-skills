# --- metade PUBLICADA: bibliotecas no registry -----------------------------
kind = library
target = @usetheo/skills-sdk @usetheo/skills-cli @usetheo/skills-mcp @usetheo/skills
build = pnpm install --frozen-lockfile && pnpm -r build
verify = npm view @usetheo/skills-cli version
credentials = NPM_TOKEN

# NPM_TOKEN vive em ~/.npmrc (escopo de usuário, fora do repo). NUNCA escreva o valor
# aqui nem em arquivo do projeto. Faltando credencial numa execução: PEÇA ao humano.

# --- metade INTERNA: o serviço, que não é publicado ------------------------
# `@usetheo/skills-api` é private=true por decisão. Para ele vale a rota interna do
# cycle-acceptance § Target kinds: build a partir da TAG DE RELEASE em checkout limpo,
# executado como o consumidor real o executa — nunca a working tree, nunca `pnpm dev`.
#
# internal_target = @usetheo/skills-api servido a partir do build da tag
# internal_build  = git worktree add /tmp/acc-m30 <tag> && cd /tmp/acc-m30 && pnpm install --frozen-lockfile && pnpm -r build && pnpm --filter @usetheo/skills-api start
# internal_deps   = postgres do próprio projeto (docker compose -f docker-compose.yaml up -d);
